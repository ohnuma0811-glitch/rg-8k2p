## Global Constraints

- 実行時に読み込むライブラリは Three.js r170（`vendor/three.module.min.js` をファイルの場所で読む。importmap は iOS 16.4 より前で動かないので使わない）と、M4 の Firebase（gstatic.com の決まった版）だけ。ビルドの道具は使わない。`package.json` はテスト用で、依存パッケージなし。
- 物理の更新は 60Hz 固定タイムステップ。描画が遅れてもゲームの速さは変えない。
- 横向きに固定。縦向きのときは「Turn your device」を表示。
- 画面の文字は英語（チャットの定型文だけ English / 日本語）。ボタンにはアイコンを付ける。
- 操作: ボタン（左下に ◀ ▶ と ▼、右下に DRIFT と ITEM）か傾き。アクセルは自動。キーボードは ← → でハンドル、↓ でブレーキ、0 でドリフト、Enter でアイテム。パソコンでは傾きを選べない（操作の選択を飛ばす）。
- 向きの決まり: `heading` は x 軸から z 軸へ回る角度で、増える方が右曲がり。`steer` と `lateral`（中心線からの距離）は右が正。
- 1周 30〜50 秒（1300m、万能タイプで約40秒）、3周。8人（足りない分は CPU）。道幅18m、ダートは片側4m。
- 絵・音のファイルは使わない（形と色と音はプログラムで作る）。`docs/reference/penguin-man.webp` は見本で、ゲームからは読まない。
- 任天堂のデザイン・曲・名前は使わない（効果はまねてよい）。
- 安い Android タブレットでも 30fps 以上（新しいスマホは 60fps が目標）。重いときは画面の細かさを自動で下げる。
- コードのコメントは日本語で簡潔に。識別子は英語。

## Review Focus

1. 横向きを反対（画面の回転 270 度）にして持つ・遊んでいる途中で回す → 傾きの左右が逆にならない（M1 Task 4 でテスト）。
2. アプリを切り替えて長くたってから戻る → 車がワープしない（1回の描画での更新は最大5回。M1 Task 1 でテスト）。
3. ◀ を押したまま指を ▶ へすべらせる・通知で `touchcancel` が来る → すぐ切り替わる・押しっぱなしにならない（M1 Task 4 でテスト）。

`src/loop.js`:

```js
// 60Hz 固定タイムステップ。描画が遅れてもゲームの速さは変えない
export const STEP = 1 / 60;

export function advance(acc, dt, step, maxSteps = 5) {
  acc += Math.min(Math.max(dt, 0), step * maxSteps);
  const steps = Math.floor(acc / step + 1e-9);
  acc = Math.max(0, acc - steps * step);
  return { steps, acc };
}

export function startLoop({ update, render, step = STEP }) {
  let last = null;
  let acc = 0;
  let running = true;
  function frame(t) {
    if (!running) return;
    const now = t / 1000;
    const dt = last === null ? 0 : now - last;
    last = now;
    const r = advance(acc, dt, step);
    acc = r.acc;
    for (let i = 0; i < r.steps; i++) update(step);
    render();
    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
  return { stop() { running = false; } };
}
```

```js
// コースのデータから1mごとの点の並びを作り、位置からコース上の場所を求める
export const SAMPLE_STEP = 1;

// 4つの点 p0〜p3 のうち、p1 から p2 までをなめらかにつなぐ線（centripetal Catmull-Rom）
function catmullRom(p0, p1, p2, p3, steps) {
  const knot = (a, b) => Math.sqrt(Math.hypot(b[0] - a[0], b[1] - a[1]));
  const t1 = knot(p0, p1);
  const t2 = t1 + knot(p1, p2);
  const t3 = t2 + knot(p2, p3);
  const mix = (a, b, ta, tb, t) => [
    ((tb - t) * a[0] + (t - ta) * b[0]) / (tb - ta),
    ((tb - t) * a[1] + (t - ta) * b[1]) / (tb - ta),
  ];
  const out = [];
  for (let k = 0; k < steps; k++) {
    const t = t1 + ((t2 - t1) * k) / steps;
    const a1 = mix(p0, p1, 0, t1, t);
    const a2 = mix(p1, p2, t1, t2, t);
    const a3 = mix(p2, p3, t2, t3, t);
    const b1 = mix(a1, a2, 0, t2, t);
    const b2 = mix(a2, a3, t1, t3, t);
    out.push(mix(b1, b2, t1, t2, t));
  }
  return out;
}

// 閉じたなめらかな線を細かい折れ線にする
function smoothLoop(points) {
  const n = points.length;
  const line = [];
  for (let i = 0; i < n; i++) {
    line.push(...catmullRom(points[(i - 1 + n) % n], points[i], points[(i + 1) % n], points[(i + 2) % n], 40));
  }
  return line;
}

// 折れ線を、決めた長さ（m）に大きくして、1m ごとに点を取り直す。真ん中が (0, 0) になるように置く
function resample(line, length) {
  const seg = line.map((p, i) => Math.hypot(line[(i + 1) % line.length][0] - p[0], line[(i + 1) % line.length][1] - p[1]));
  const raw = seg.reduce((a, b) => a + b, 0);
  const k = length / raw;
  const xs = line.map((p) => p[0]);
  const zs = line.map((p) => p[1]);
  const cx = (Math.min(...xs) + Math.max(...xs)) / 2;
  const cz = (Math.min(...zs) + Math.max(...zs)) / 2;
  const n = Math.round(length / SAMPLE_STEP);
  const out = [];
  let j = 0;
  let acc = 0;
  for (let i = 0; i < n; i++) {
    const want = ((i * length) / n) / k;
    while (acc + seg[j] < want) {
      acc += seg[j];
      j++;
    }
    const a = line[j];
    const b = line[(j + 1) % line.length];
    const t = seg[j] > 0 ? (want - acc) / seg[j] : 0;
    out.push({ x: (a[0] + (b[0] - a[0]) * t - cx) * k, z: (a[1] + (b[1] - a[1]) * t - cz) * k, s: (i * length) / n });
  }
  return out;
}

// 中心線が自分と交わる所を探す（となり合う部分は除く）
function findCrossings(samples) {
  const n = samples.length;
  const found = [];
  for (let i = 0; i < n; i++) {
    const a = samples[i];
    const b = samples[(i + 1) % n];
    for (let j = i + 50; j < n; j++) {
      if (n - (j - i) < 50) continue;
      const c = samples[j];
      const d = samples[(j + 1) % n];
      const den = (b.x - a.x) * (d.z - c.z) - (b.z - a.z) * (d.x - c.x);
      if (den === 0) continue;
      const t = ((c.x - a.x) * (d.z - c.z) - (c.z - a.z) * (d.x - c.x)) / den;
      const u = ((c.x - a.x) * (b.z - a.z) - (c.z - a.z) * (b.x - a.x)) / den;
      if (t >= 0 && t < 1 && u >= 0 && u < 1) {
        found.push({ x: a.x + (b.x - a.x) * t, z: a.z + (b.z - a.z) * t, sA: a.s + t * SAMPLE_STEP, sB: c.s + u * SAMPLE_STEP, iA: i, iB: j });
      }
    }
  }
  return found;
}

function inRange(s, from, to) {
  return s >= from && s < to;
}

export function buildTrack(def) {
  const samples = resample(smoothLoop(def.points), def.length);
  const n = samples.length;
  for (let i = 0; i < n; i++) {
    const p = samples[i];
    const q = samples[(i + 1) % n];
    p.h = Math.atan2(q.z - p.z, q.x - p.x);
  }
  // ジャンプがあるコースは、交わる所がちょうど1か所
  let cross = null;
  if (def.jump) {
    const crossings = findCrossings(samples);
    if (crossings.length !== 1) throw new Error(`コースの交わる所は1か所のはず: ${crossings.length}`);
    cross = crossings[0];
  }
  const j = def.jump;
  const at = (d) => cross.sA + d;
  const fullWall = def.roadHalf + def.dirt;
  for (const p of samples) {
    p.y = 0;
    p.wall = fullWall;
    p.ramp = false;
    p.boost = false;
    p.noWall = false;
    if (!cross) continue;
    p.boost = inRange(p.s, at(j.boost.from), at(j.boost.to));
    if (inRange(p.s, at(j.ramp.from), at(j.ramp.to))) {
      p.ramp = true;
      p.y = (j.ramp.height * (p.s - at(j.ramp.from))) / (j.ramp.to - j.ramp.from);
      p.wall = def.roadHalf;
    }
    if (inRange(p.s, at(j.gap.from), at(j.gap.to))) {
      p.y = null;
      p.wall = def.roadHalf;
    }
    const dx = p.x - cross.x;
    const dz = p.z - cross.z;
    p.noWall = p.y === 0 && dx * dx + dz * dz < def.noWallRadius ** 2;
  }
  const startS = cross ? cross.sB - def.startBeforeCross : 0;
  const startIndex = samples.findIndex((p) => p.s >= startS);
  return { def, samples, length: def.length, roadHalf: def.roadHalf, startIndex, cross };
}

// 位置 (x, z) に一番近いコースの点を探す。hint（前の点の番号）があれば、その前後だけを探す
// （交わる所で、もう一方の道の点を拾わないため）
export function locate(track, x, z, hint = null, win = 40) {
  const { samples, length, roadHalf } = track;
  const n = samples.length;
  let best = 0;
  let bestD = Infinity;
  const scan = (i) => {
    const p = samples[i];
    const dx = x - p.x;
    const dz = z - p.z;
    const d = dx * dx + dz * dz;
    if (d < bestD) {
      bestD = d;
      best = i;
    }
  };
  if (hint === null) {
    for (let i = 0; i < n; i++) scan(i);
  } else {
    for (let k = -win; k <= win; k++) scan((((hint + k) % n) + n) % n);
  }
  const p = samples[best];
  const fx = Math.cos(p.h);
  const fz = Math.sin(p.h);
  const dx = x - p.x;
  const dz = z - p.z;
  // 右向きの距離（右が正）
  const lateral = dx * -fz + dz * fx;
  const s = ((((p.s + dx * fx + dz * fz) % length) + length) % length);
  const a = Math.abs(lateral);
  const surface = a <= roadHalf ? 'road' : a <= p.wall ? 'dirt' : 'out';
  return {
    i: best,
    s,
    lateral,
    heading: p.h,
    ground: p.y,
    wall: p.wall,
    surface,
    boost: p.boost && a <= roadHalf,
  };
}

// 落ちた所 i から戻って、平らな地面が10m続いた所を、戻る場所にする
export function respawnIndex(track, i) {
  const { samples } = track;
  const n = samples.length;
  let run = 0;
  for (let k = 1; k < n; k++) {
    const j = (((i - k) % n) + n) % n;
    const p = samples[j];
    const flat = p.y === 0 && !p.ramp && !p.boost;
    run = flat ? run + 1 : 0;
    if (run >= 10) return j;
  }
  return track.startIndex;
}
```

```js
// 車の動き。アクセルは自動。ハンドル・ブレーキ・ドリフト・ダート・柵・ジャンプ台
import { locate, respawnIndex } from './track.js';

export const PHYS = {
  DIRT_FACTOR: 0.55, // ダートでの最高速の割合
  DRAG: 12, // 最高速より速いときに落ちる速さ（m/s²）
  BRAKE: 25, // ブレーキ（m/s²）
  REVERSE_MAX: 6, // バックの最高速（m/s）
  TURN_FULL_SPEED: 6, // この速さ以上で、曲がる力がいっぱいになる（止まっていると曲がれない）
  DRIFT_MIN_SPEED: 12, // ドリフトできる速さ
  DRIFT_BASE: 0.35, // ドリフト中、ハンドルを切っていなくても曲がる割合
  DRIFT_ADJ: 0.3, // ドリフト中、ハンドルで足し引きできる割合
  BLUE_TIME: 0.9, // 青い火花になるまでの時間（秒）
  ORANGE_TIME: 1.9, // オレンジの火花になるまでの時間（秒）
  BLUE_BOOST: 0.6, // 青で離したときの加速の時間（秒）
  ORANGE_BOOST: 1.2, // オレンジで離したときの加速の時間（秒）
  PAD_BOOST: 1.0, // 加速板の加速の時間（秒）
  BOOST_FACTOR: 1.3, // 加速中の最高速の倍率
  GRAVITY: 15, // m/s²
  KICK: (18 * Math.PI) / 180, // ジャンプ台から飛び出す角度
  LAUNCH_SPEED: 30, // ジャンプ台から飛び出す速さ（いつも同じ。必ず向こう側の同じ所に着地する）
  AIR_TURN: 0.3, // 空中で曲がれる割合
  WALL_KEEP: 0.6, // 柵にぶつかったときに残る速さの割合
  FALL_Y: -2, // これより下に落ちたらコースに戻す
  RESPAWN_DELAY: 2, // コースに戻すまでの時間（秒）
};

function wrapAngle(a) {
  while (a > Math.PI) a -= 2 * Math.PI;
  while (a < -Math.PI) a += 2 * Math.PI;
  return a;
}

export function placeAt(kart, track, i, lateral = 0) {
  const p = track.samples[i];
  kart.x = p.x + -Math.sin(p.h) * lateral;
  kart.z = p.z + Math.cos(p.h) * lateral;
  kart.y = p.y ?? 0;
  kart.vy = 0;
  kart.heading = p.h;
  kart.speed = 0;
  kart.seg = i;
  kart.airborne = false;
  kart.drift = { active: false, dir: 0, charge: 0 };
  kart.boost = 0;
  kart.surface = 'road';
  kart.respawn = 0;
  kart.respawnAt = i;
  kart.prevGround = kart.y;
}

export function createKart(track, i, lateral = 0) {
  const kart = {};
  placeAt(kart, track, i, lateral);
  return kart;
}

// 0 = 火花なし、1 = 青、2 = オレンジ
export function driftLevel(kart) {
  if (!kart.drift.active) return 0;
  if (kart.drift.charge >= PHYS.ORANGE_TIME) return 2;
  if (kart.drift.charge >= PHYS.BLUE_TIME) return 1;
  return 0;
}

function startBoost(kart, params, time) {
  kart.boost = Math.max(kart.boost, time);
  kart.speed = Math.max(kart.speed, params.topSpeed * PHYS.BOOST_FACTOR);
}

function endDrift(kart, params, release) {
  const lv = driftLevel(kart);
  kart.drift = { active: false, dir: 0, charge: 0 };
  if (release && lv === 2) startBoost(kart, params, PHYS.ORANGE_BOOST);
  else if (release && lv === 1) startBoost(kart, params, PHYS.BLUE_BOOST);
}

// ctl: { steer: -1〜1（右が正）, brake: bool, drift: bool }
export function stepKart(kart, ctl, track, params, dt) {
  if (kart.respawn > 0) {
    kart.respawn -= dt;
    if (kart.respawn <= 0) placeAt(kart, track, kart.respawnAt);
    return;
  }
  const here = locate(track, kart.x, kart.z, kart.seg);
  const steer = Math.max(-1, Math.min(1, ctl.steer));

  if (!kart.airborne) {
    if (here.boost) startBoost(kart, params, PHYS.PAD_BOOST);
    let maxV = params.topSpeed;
    if (here.surface !== 'road') maxV *= PHYS.DIRT_FACTOR;
    if (kart.boost > 0) maxV *= PHYS.BOOST_FACTOR;
    if (ctl.brake) {
      kart.speed = Math.max(-PHYS.REVERSE_MAX, kart.speed - PHYS.BRAKE * dt);
    } else if (kart.speed < maxV) {
      kart.speed = Math.min(maxV, kart.speed + params.accel * dt);
    } else {
      kart.speed = Math.max(maxV, kart.speed - PHYS.DRAG * dt);
    }

    // ドリフト: 押したときにハンドルを切っている向きに曲がり続ける。離すと、たまった分だけ加速
    if (!kart.drift.active && ctl.drift && Math.abs(steer) > 0.3 && kart.speed >= PHYS.DRIFT_MIN_SPEED) {
      kart.drift = { active: true, dir: Math.sign(steer), charge: 0 };
    } else if (kart.drift.active) {
      if (!ctl.drift) endDrift(kart, params, true);
      else if (kart.speed < PHYS.DRIFT_MIN_SPEED) endDrift(kart, params, false);
      else kart.drift.charge += dt;
    }

    const grip = Math.min(1, Math.abs(kart.speed) / PHYS.TURN_FULL_SPEED) * Math.sign(kart.speed);
    const turn = kart.drift.active
      ? kart.drift.dir * PHYS.DRIFT_BASE + steer * PHYS.DRIFT_ADJ
      : steer;
    kart.heading += turn * params.turnRate * grip * dt;
  } else {
    kart.heading += steer * params.turnRate * PHYS.AIR_TURN * dt;
  }
  kart.heading = wrapAngle(kart.heading);
  kart.boost = Math.max(0, kart.boost - dt);

  kart.x += Math.cos(kart.heading) * kart.speed * dt;
  kart.z += Math.sin(kart.heading) * kart.speed * dt;
  const loc = locate(track, kart.x, kart.z, kart.seg);
  kart.seg = loc.i;
  kart.surface = loc.surface;

  if (kart.airborne) {
    kart.vy -= PHYS.GRAVITY * dt;
    kart.y += kart.vy * dt;
    if (loc.ground !== null && kart.y <= loc.ground) {
      kart.y = loc.ground;
      kart.vy = 0;
      kart.airborne = false;
    } else if (kart.y < PHYS.FALL_Y) {
      kart.respawn = PHYS.RESPAWN_DELAY;
      kart.respawnAt = respawnIndex(track, loc.i);
      kart.speed = 0;
      return;
    }
  } else if (loc.ground === null) {
    // 地面がなくなった。前に走っていればジャンプ台から飛び出す（いつも同じ速さで、必ず向こう側に届く）
    kart.airborne = true;
    if (kart.drift.active) endDrift(kart, params, false);
    if (kart.speed > 0) {
      kart.speed = PHYS.LAUNCH_SPEED;
      kart.vy = kart.speed * Math.sin(PHYS.KICK);
    } else {
      kart.vy = 0;
    }
  } else {
    kart.y = loc.ground;
  }

  // 柵: 外に出ようとしたら押し戻し、コースの向きにそろえて、スピードを落とす
  const a = Math.abs(loc.lateral);
  if (a > loc.wall) {
    const side = Math.sign(loc.lateral);
    const excess = loc.lateral - side * loc.wall;
    kart.x -= -Math.sin(loc.heading) * excess;
    kart.z -= Math.cos(loc.heading) * excess;
    const out = Math.sin(wrapAngle(kart.heading - loc.heading)) * side * Math.sign(kart.speed);
    if (out > 0) {
      kart.heading = loc.heading;
      kart.speed *= PHYS.WALL_KEEP;
    }
  }
}
```

```js
// タッチボタン・傾き・キーボードを、steer（-1〜1、右が正）/ brake / drift / item にまとめる
const KEYS = {
  left: ['ArrowLeft'],
  right: ['ArrowRight'],
  brake: ['ArrowDown'],
  drift: ['Digit0', 'Numpad0'],
  item: ['Enter', 'NumpadEnter'],
};
const HANDLED = new Set(Object.values(KEYS).flat());

// mode: 'buttons'（◀ ▶ を出す）/ 'tilt'（◀ ▶ を出さない）/ 'keyboard'（ボタンを出さない）
export function computeButtonLayout(winW, winH, safe = { left: 0, right: 0, bottom: 0 }, mode = 'buttons') {
  // 半径は画面の短い辺の9%、34〜56px（約1〜1.5cm）
  const r = Math.max(34, Math.min(56, Math.min(winW, winH) * 0.09));
  const margin = 16;
  const y = winH - safe.bottom - margin - r;
  const left = { x: safe.left + 8 + r, y, r };
  const right = { x: left.x + r * 2 + 16, y, r };
  const brakeR = r * 0.8;
  // ブレーキは ◀ ▶ の間の上。傾きのときは ◀ の場所
  const brake = mode === 'tilt'
    ? { x: left.x, y, r: brakeR }
    : { x: (left.x + right.x) / 2, y: y - r - 12 - brakeR, r: brakeR };
  const driftR = r * 1.25;
  const drift = { x: winW - safe.right - margin - driftR, y: winH - safe.bottom - margin - driftR, r: driftR };
  const itemR = r * 0.85;
  const item = { x: drift.x, y: drift.y - driftR - 12 - itemR, r: itemR };
  return { mode, left, right, brake, drift, item, splitX: (left.x + right.x) / 2, winW, winH };
}

function hit(p, b, scale) {
  return Math.hypot(p.x - b.x, p.y - b.y) <= b.r * scale;
}

export function resolveTouches(points, layout) {
  const s = { left: false, right: false, brake: false, drift: false, item: false };
  if (layout.mode === 'keyboard') return s;
  for (const p of points) {
    if (p.x < layout.winW / 2) {
      if (layout.mode === 'tilt') {
        if (hit(p, layout.brake, 1.6)) s.brake = true;
      } else if (hit(p, layout.brake, 1.3)) {
        s.brake = true;
      } else if (p.y > layout.brake.y + layout.brake.r) {
        // ◀ ▶ の段は、指をすべらせても左右が切り替わるよう、真ん中で分ける
        if (p.x < layout.splitX) s.left = true;
        else s.right = true;
      }
    } else if (hit(p, layout.item, 1.3)) {
      s.item = true;
    } else if (p.y > layout.winH * 0.35) {
      s.drift = true;
    }
  }
  return s;
}

// 傾きセンサー（重力をふくむ加速度、端末の向きの x・y）から steer を求める
// screenAngle は画面の回転（0 / 90 / 180 / 270）。invert は符号が逆に出る端末（iPhone）用
export const TILT_DEAD = 3; // これより小さい傾きは0（度）
export const TILT_FULL = 25; // この傾きでハンドルいっぱい（度）
export function tiltSteer(ax, ay, screenAngle, invert = false) {
  const a = (screenAngle * Math.PI) / 180;
  // 画面の右向きの成分
  let sx = ax * Math.cos(a) - ay * Math.sin(a);
  if (invert) sx = -sx;
  const deg = (Math.asin(Math.max(-1, Math.min(1, -sx / 9.81))) * 180) / Math.PI;
  const mag = Math.abs(deg);
  if (mag < TILT_DEAD) return 0;
  return Math.sign(deg) * Math.min(1, (mag - TILT_DEAD) / (TILT_FULL - TILT_DEAD));
}

export class Controls {
  constructor() {
    this.mode = 'buttons';
    this.touch = resolveTouches([], { mode: 'keyboard' });
    this.points = [];
    this.keys = new Set();
    this.tilt = 0;
    this.steer = 0;
    this.brake = false;
    this.drift = false;
    this.item = false;
    this.itemPressed = false;
    this.prevItem = false;
    // 更新の間に押して離されたアイテムボタンも拾うための記録
    this.itemLatch = false;
  }

  setMode(mode) {
    this.mode = mode;
    this.tilt = 0;
  }

  setTouches(points, layout) {
    const was = this.touch.item;
    this.points = points;
    this.touch = resolveTouches(points, layout);
    if (this.touch.item && !was) this.itemLatch = true;
  }

  // 画面サイズが変わったとき、押している指を新しい配置で判定し直す
  relayout(layout) {
    this.touch = resolveTouches(this.points, layout);
  }

  setTilt(value) {
    this.tilt = value;
  }

  keyDown(code, repeat = false) {
    if (KEYS.item.includes(code) && !repeat) this.itemLatch = true;
    this.keys.add(code);
  }

  keyUp(code) {
    this.keys.delete(code);
  }

  clear() {
    this.touch = resolveTouches([], { mode: 'keyboard' });
    this.points = [];
    this.keys.clear();
    this.itemLatch = false;
  }

  // 更新1回につき1度だけ呼ぶ
  tick() {
    const has = (name) => KEYS[name].some((k) => this.keys.has(k));
    const left = this.touch.left || has('left');
    const right = this.touch.right || has('right');
    const keyOrButton = (right ? 1 : 0) - (left ? 1 : 0);
    this.steer = keyOrButton !== 0 || this.mode !== 'tilt' ? keyOrButton : this.tilt;
    this.brake = this.touch.brake || has('brake');
    this.drift = this.touch.drift || has('drift');
    this.item = this.touch.item || has('item');
    this.itemPressed = (this.item && !this.prevItem) || this.itemLatch;
    this.itemLatch = false;
    this.prevItem = this.item;
  }
}

export function attachControls(controls, target, getLayout) {
  const onTouch = (e) => {
    e.preventDefault();
    const points = Array.from(e.touches, (t) => ({ x: t.clientX, y: t.clientY }));
    controls.setTouches(points, getLayout());
  };
  for (const type of ['touchstart', 'touchmove', 'touchend', 'touchcancel']) {
    target.addEventListener(type, onTouch, { passive: false });
  }
  window.addEventListener('keydown', (e) => {
    if (HANDLED.has(e.code)) {
      e.preventDefault();
      controls.keyDown(e.code, e.repeat);
    }
  });
  window.addEventListener('keyup', (e) => controls.keyUp(e.code));
  window.addEventListener('blur', () => controls.clear());
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) controls.clear();
  });
}

// 傾き操作ができる端末か（タッチと傾きセンサーがある）
export function canTilt() {
  return typeof window !== 'undefined'
    && (navigator.maxTouchPoints > 0 || 'ontouchstart' in window)
    && 'DeviceMotionEvent' in window;
}

// iPhone・iPad は許可が必要。ボタンを押したときに呼ぶ。許可されたら true
export async function requestTiltPermission() {
  const DME = window.DeviceMotionEvent;
  if (DME && typeof DME.requestPermission === 'function') {
    try {
      return (await DME.requestPermission()) === 'granted';
    } catch {
      return false;
    }
  }
  return true;
}

function screenAngle() {
  const a = screen.orientation?.angle ?? window.orientation ?? 0;
  return ((a % 360) + 360) % 360;
}

export function startTilt(controls) {
  // iPhone は重力の向きが Android と逆に出る
  const invert = typeof window.DeviceMotionEvent?.requestPermission === 'function';
  window.addEventListener('devicemotion', (e) => {
    const g = e.accelerationIncludingGravity;
    if (!g || g.x === null || g.y === null) return;
    controls.setTilt(tiltSteer(g.x, g.y, screenAngle(), invert));
  });
}
```

```js
// 画面の細かさ（0.5〜1）を、なめらかさ（fps）に合わせて自動で上げ下げする
export const QUALITY_MIN = 0.5;
export const QUALITY_MAX = 1;

export function nextQuality(q, fps) {
  if (fps < 45) return Math.max(QUALITY_MIN, Math.round((q - 0.15) * 100) / 100);
  if (fps > 57) return Math.min(QUALITY_MAX, Math.round((q + 0.05) * 100) / 100);
  return q;
}

// 実際に描く細かさ（端末の画素の細かさ。上限2）
export function pixelRatio(devicePixelRatio, q) {
  return Math.min(devicePixelRatio || 1, 2) * q;
}
```

```js
  ground.rotation.x = -Math.PI / 2;
  scene.add(ground);

// 車の後ろの低い所から追いかける。速いほど視野を広げ、加速中はさらに広げる（スピード感）
// 急に動かないよう、少しずつ近づける
export function updateChaseCamera(camera, kart, topSpeed, dt) {
  const fx = Math.cos(kart.heading);
  const fz = Math.sin(kart.heading);
  const want = new THREE.Vector3(kart.x - fx * 6, kart.y + 2.6, kart.z - fz * 6);
  const look = new THREE.Vector3(kart.x + fx * 6, kart.y + 1, kart.z + fz * 6);
  const st = camera.userData;
  if (!st.pos) {
    st.pos = want.clone();
    st.look = look.clone();
    st.fov = camera.fov;
  }
  const k = 1 - Math.exp(-dt * 8);
  st.pos.lerp(want, k);
  st.look.lerp(look, k);
  const fast = Math.max(0, Math.min(1, (kart.speed - topSpeed * 0.5) / (topSpeed * 0.5)));
  const fov = 68 + fast * 12 + (kart.boost > 0 ? 8 : 0);
  st.fov += (fov - st.fov) * (1 - Math.exp(-dt * 4));
  camera.fov = st.fov;
  camera.updateProjectionMatrix();
  camera.position.copy(st.pos);
  camera.lookAt(st.look);
}
```

```js
import * as THREE from 'three';

const COLORS = {
  dirt: 0xb08850,
  rampSide: 0x8a6d4a,
  red: 0xd8262b,
  white: 0xffffff,
};

// 右向きのベクトル（右が正）
function rightOf(p) {
  return [-Math.sin(p.h), Math.cos(p.h)];
}

function mesh(pos, { color, colors, uv, map } = {}) {
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  if (colors) g.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
  if (uv) g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
  g.computeVertexNormals();
  const m = new THREE.MeshLambertMaterial({ color, map, vertexColors: Boolean(colors), side: THREE.DoubleSide });
  return new THREE.Mesh(g, m);
}

// となり合う点どうしを四角でつないだ、地面に平らな帯。ok(p) が false の点はつながない
// colorOf があれば四角ごとの色、vScale があれば模様を貼る（縦は vScale m ごとにくり返す）
function ribbon(samples, offA, offB, yOf, ok, { color, colorOf, map, vScale } = {}) {
  const pos = [];
  const colors = colorOf ? [] : null;
  const uv = map ? [] : null;
  const c = new THREE.Color();
  const n = samples.length;
  for (let i = 0; i < n; i++) {
    const p = samples[i];
    const q = samples[(i + 1) % n];
    if (!ok(p) || !ok(q)) continue;
    const [rpx, rpz] = rightOf(p);
    const [rqx, rqz] = rightOf(q);
    const a = [p.x + rpx * offA, yOf(p), p.z + rpz * offA];
    const b = [p.x + rpx * offB, yOf(p), p.z + rpz * offB];
    const cc = [q.x + rqx * offA, yOf(q), q.z + rqz * offA];
    const d = [q.x + rqx * offB, yOf(q), q.z + rqz * offB];
    pos.push(...a, ...cc, ...b, ...b, ...cc, ...d);
    if (colors) {
      c.setHex(colorOf(p));
      for (let k = 0; k < 6; k++) colors.push(c.r, c.g, c.b);
    }
    if (uv) {
      const v0 = p.s / vScale;
      const v1 = v0 + (q.s > p.s ? q.s - p.s : 1) / vScale;
      uv.push(0, v0, 0, v1, 1, v0, 1, v0, 0, v1, 1, v1);
    }
  }
  return mesh(pos, { color, colors, uv, map });
}

// 道の横に立つ縦の板（柵やジャンプ台の側面）
function fence(samples, off, yBottom, yTop, ok, colorOf) {
  const pos = [];
  const col = [];
  const n = samples.length;
  const c = new THREE.Color();
  for (let i = 0; i < n; i++) {
    const p = samples[i];
    const q = samples[(i + 1) % n];
    if (!ok(p) || !ok(q)) continue;
    const [rpx, rpz] = rightOf(p);
    const [rqx, rqz] = rightOf(q);
    const px = p.x + rpx * off;
    const pz = p.z + rpz * off;
    const qx = q.x + rqx * off;
    const qz = q.z + rqz * off;
    pos.push(px, yBottom(p), pz, qx, yBottom(q), qz, px, yTop(p), pz);
    pos.push(px, yTop(p), pz, qx, yBottom(q), qz, qx, yTop(q), qz);
    c.setHex(colorOf(p));
    for (let k = 0; k < 6; k++) col.push(c.r, c.g, c.b);
  }
  return mesh(pos, { colors: col });
}

function canvasTexture(w, h, draw) {
  const cv = document.createElement('canvas');
  cv.width = w;
  cv.height = h;
  draw(cv.getContext('2d'), w, h);
  const t = new THREE.CanvasTexture(cv);
  t.colorSpace = THREE.SRGBColorSpace;
  t.wrapT = THREE.RepeatWrapping;
  t.anisotropy = 4;
  return t;
}

// コースを横切る板（スタートライン、加速板）。from〜to は s（m）
function strip(track, from, to, y, material) {
  const { samples, roadHalf } = track;
  const pos = [];
  const uv = [];
  const list = samples.filter((p) => p.s >= from && p.s < to);
  for (let k = 0; k + 1 < list.length; k++) {
    const p = list[k];
    const q = list[k + 1];
    const [rpx, rpz] = rightOf(p);
    const [rqx, rqz] = rightOf(q);
    const v0 = k / (list.length - 1);
    const v1 = (k + 1) / (list.length - 1);
    const a = [p.x - rpx * roadHalf, y(p), p.z - rpz * roadHalf];
    const b = [p.x + rpx * roadHalf, y(p), p.z + rpz * roadHalf];
    const c = [q.x - rqx * roadHalf, y(q), q.z - rqz * roadHalf];
    const d = [q.x + rqx * roadHalf, y(q), q.z + rqz * roadHalf];
    pos.push(...a, ...c, ...b, ...b, ...c, ...d);
    uv.push(0, v0, 0, v1, 1, v0, 1, v0, 0, v1, 1, v1);
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
  return new THREE.Mesh(g, material);
}

// 点 (x, z) がコースの線で囲まれた内側にあるか（線を横切る回数が奇数なら内側）
function insideLoop(samples, x, z) {
  let inside = false;
  for (let i = 0, j = samples.length - 1; i < samples.length; j = i++) {
    const a = samples[i];
    const b = samples[j];
    if (a.z > z !== b.z > z && x < ((b.x - a.x) * (z - a.z)) / (b.z - a.z) + a.x) inside = !inside;
  }
  return inside;
}

// 点 (x, z) からコースの中心線までの一番近い距離
function distToTrack(samples, x, z) {
  let d = Infinity;
  for (const p of samples) d = Math.min(d, (p.x - x) ** 2 + (p.z - z) ** 2);
  return Math.sqrt(d);
}

// スタートラインの横（コースの外側、柵の外）に、段になった観客席と観客
function grandstand(track) {
  const group = new THREE.Group();
  const p = track.samples[track.startIndex];
  const base = p.wall + 4;
  const [r0x, r0z] = rightOf(p);
  const side = insideLoop(track.samples, p.x + r0x * (base + 6), p.z + r0z * (base + 6)) ? -1 : 1;
  const rx = r0x * side;
  const rz = r0z * side;
  const length = 50;
  const steps = 6;
  const stand = new THREE.MeshLambertMaterial({ color: 0x9aa3ad });
  for (let k = 0; k < steps; k++) {
    const box = new THREE.Mesh(new THREE.BoxGeometry(length, 1 + k * 1.2, 2), stand);
    const d = base + k * 2;
    box.position.set(p.x + rx * d, (1 + k * 1.2) / 2, p.z + rz * d);
    box.rotation.y = -p.h;
    group.add(box);
  }
  const perRow = 40;
  const people = new THREE.InstancedMesh(new THREE.BoxGeometry(0.7, 1, 0.6), new THREE.MeshLambertMaterial(), steps * perRow);
  const m = new THREE.Matrix4();
  const c = new THREE.Color();
  const palette = [0xe63946, 0xf1c453, 0x2a9d8f, 0x457b9d, 0xffffff, 0xf4a261, 0x9b5de5];
  let idx = 0;
  for (let k = 0; k < steps; k++) {
    for (let j = 0; j < perRow; j++) {
      const along = -length / 2 + 0.6 + j * ((length - 1.2) / (perRow - 1));
      const d = base + k * 2;
      const x = p.x + rx * d + Math.cos(p.h) * along;
      const z = p.z + rz * d + Math.sin(p.h) * along;
      m.makeTranslation(x, 1 + k * 1.2 + 0.5, z);
      people.setMatrixAt(idx, m);
      people.setColorAt(idx, c.setHex(palette[(k * 7 + j * 3) % palette.length]));
      idx++;
    }
  }
  group.add(people);
  return group;
}

// 柵の外に、木と看板を並べる（流れていく物があると速く感じる）。ほかの道や観客席には置かない
function roadside(track) {
  const { samples } = track;
  const start = samples[track.startIndex];
  const trees = [];
  const signs = [];
  for (let i = 0; i < samples.length; i += 9) {
    const p = samples[i];
    if (p.y !== 0) continue;
    const [rx, rz] = rightOf(p);
    for (const side of [-1, 1]) {
      const d = p.wall + 3 + ((i * 7) % 5);
      const x = p.x + rx * side * d;
      const z = p.z + rz * side * d;
      if (distToTrack(samples, x, z) < p.wall + 2.5) continue;
      if (Math.hypot(x - start.x, z - start.z) < 45) continue;
      if ((i / 9 + (side > 0 ? 1 : 0)) % 4 === 0) signs.push({ x, z, h: p.h });
      else trees.push({ x, z });
    }
  }
  const group = new THREE.Group();
  const m = new THREE.Matrix4();
  const trunk = new THREE.InstancedMesh(new THREE.CylinderGeometry(0.3, 0.4, 2, 6), new THREE.MeshLambertMaterial({ color: 0x7a4b25 }), trees.length);
  const leaves = new THREE.InstancedMesh(new THREE.ConeGeometry(2.2, 5, 7), new THREE.MeshLambertMaterial({ color: 0x2f7d32 }), trees.length);
  trees.forEach((t, k) => {
    trunk.setMatrixAt(k, m.makeTranslation(t.x, 1, t.z));
    leaves.setMatrixAt(k, m.makeTranslation(t.x, 4.2, t.z));
  });
  const board = new THREE.InstancedMesh(new THREE.BoxGeometry(4, 2, 0.2), new THREE.MeshLambertMaterial(), signs.length);
  const pole = new THREE.InstancedMesh(new THREE.CylinderGeometry(0.12, 0.12, 2.5, 5), new THREE.MeshLambertMaterial({ color: 0xdddddd }), signs.length);
  const c = new THREE.Color();
  const palette = [0xffd84a, 0x3fa9ff, 0xff5a5f, 0x7bd389];
  const q = new THREE.Quaternion();
  const up = new THREE.Vector3(0, 1, 0);
  signs.forEach((s, k) => {
    q.setFromAxisAngle(up, -s.h);
    board.setMatrixAt(k, m.compose(new THREE.Vector3(s.x, 3.3, s.z), q, new THREE.Vector3(1, 1, 1)));
    board.setColorAt(k, c.setHex(palette[k % palette.length]));
    pole.setMatrixAt(k, m.makeTranslation(s.x, 1.25, s.z));
  });
  group.add(trunk, leaves, board, pole);
  return group;
}

export function buildTrackMesh(track) {
  const { samples, roadHalf } = track;
  const dirt = track.def.dirt;
  const group = new THREE.Group();
  const hasGround = (p) => p.y !== null;
  const flat = (p) => p.y === 0 && !p.ramp;
  const stripes = (len) => (p) => (Math.floor(p.s / len) % 2 ? COLORS.white : COLORS.red);

  // 道: 灰色に、真ん中の白い破線と、両はしの白線
  const asphalt = canvasTexture(64, 64, (g, w, h) => {
    g.fillStyle = '#55585e';
    g.fillRect(0, 0, w, h);
    g.fillStyle = '#e8e8e8';
    g.fillRect(w / 2 - 1, 0, 2, h / 2);
    g.fillRect(2, 0, 2, h);
    g.fillRect(w - 4, 0, 2, h);
  });
  group.add(ribbon(samples, -roadHalf, roadHalf, (p) => p.y + 0.03, hasGround, { color: 0xffffff, map: asphalt, vScale: 8 }));
  // 縁石（赤白）
  group.add(ribbon(samples, -roadHalf - 1, -roadHalf, (p) => p.y + 0.04, hasGround, { colorOf: stripes(3) }));
  group.add(ribbon(samples, roadHalf, roadHalf + 1, (p) => p.y + 0.04, hasGround, { colorOf: stripes(3) }));
  group.add(ribbon(samples, -roadHalf - dirt, -roadHalf - 1, () => 0.02, flat, { color: COLORS.dirt }));
  group.add(ribbon(samples, roadHalf + 1, roadHalf + dirt, () => 0.02, flat, { color: COLORS.dirt }));

  // 柵（交わる所のまわりは開ける）。ジャンプ台は道のすぐ横に側面と柵
  const fenceOk = (p) => p.y === 0 && !p.noWall && !p.ramp;
  for (const side of [-1, 1]) {
    group.add(fence(samples, side * (roadHalf + dirt), () => 0, () => 1, fenceOk, stripes(4)));
    group.add(fence(samples, side * roadHalf, () => 0, (p) => p.y + 0.03, (p) => p.ramp, () => COLORS.rampSide));
    group.add(fence(samples, side * roadHalf, (p) => p.y, (p) => p.y + 1, (p) => p.ramp, stripes(4)));
  }

  const checker = canvasTexture(64, 16, (g, w, h) => {
    for (let x = 0; x < w; x += 8) {
      for (let y = 0; y < h; y += 8) {
        g.fillStyle = (x + y) % 16 ? '#111' : '#fff';
        g.fillRect(x, y, 8, 8);
      }
    }
  });
  const startS = samples[track.startIndex].s;
  group.add(strip(track, startS - 1, startS + 2, () => 0.05, new THREE.MeshBasicMaterial({ map: checker })));

  if (track.cross) {
    const arrows = canvasTexture(64, 64, (g, w, h) => {
      g.fillStyle = '#ff8a00';
      g.fillRect(0, 0, w, h);
      g.fillStyle = '#fff36b';
      for (let k = 0; k < 2; k++) {
        const y0 = k * 32;
        g.beginPath();
        g.moveTo(8, y0 + 4);
        g.lineTo(w / 2, y0 + 28);
        g.lineTo(w - 8, y0 + 4);
        g.lineTo(w - 20, y0 + 4);
        g.lineTo(w / 2, y0 + 18);
        g.lineTo(20, y0 + 4);
        g.fill();
      }
    });
    const b = track.def.jump.boost;
    group.add(strip(track, track.cross.sA + b.from, track.cross.sA + b.to, () => 0.06, new THREE.MeshBasicMaterial({ map: arrows })));
  }

  group.add(grandstand(track));
  group.add(roadside(track));
  return group;
}
```

```js
    const w = new THREE.Mesh(wheelGeo, mat(0x111111));

export function syncKartModel(model, kart) {
  model.position.set(kart.x, kart.y, kart.z);
  model.rotation.y = -kart.heading;
  // ドリフト中は少し横を向く
  model.rotation.y -= kart.drift.active ? kart.drift.dir * 0.35 : 0;
  model.visible = kart.respawn <= 0 || Math.floor(kart.respawn * 8) % 2 === 0;
  for (const w of model.userData.wheels) w.rotation.z -= kart.speed * 0.05;
}
```

```js
// ドリフトの火花（青・オレンジ）と、ダートの砂ぼこり
import * as THREE from 'three';
import { driftLevel } from '../game/kart.js';

const MAX = 200;
const SPARK = [null, new THREE.Color(0x3fa9ff), new THREE.Color(0xff8a00)];
const DUST = new THREE.Color(0xd9b98a);

export function createEffects(scene) {
  const pos = new Float32Array(MAX * 3);
  const col = new Float32Array(MAX * 3);
  const parts = Array.from({ length: MAX }, () => ({ life: 0, x: 0, y: -100, z: 0, vx: 0, vy: 0, vz: 0, c: DUST }));
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
  geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
  const points = new THREE.Points(geo, new THREE.PointsMaterial({ size: 0.5, vertexColors: true }));
  points.frustumCulled = false;
  scene.add(points);
  let next = 0;

  function emit(x, y, z, vx, vy, vz, c, life) {
    const p = parts[next];
    next = (next + 1) % MAX;
    Object.assign(p, { x, y, z, vx, vy, vz, c, life });
  }

  return {
    update(kart, dt) {
      const fx = Math.cos(kart.heading);
      const fz = Math.sin(kart.heading);
      const rx = -fz;
      const rz = fx;
      const lv = driftLevel(kart);
      if (lv > 0) {
        for (const side of [-1, 1]) {
          emit(kart.x - fx * 1 + rx * side * 0.8, kart.y + 0.2, kart.z - fz * 1 + rz * side * 0.8,
            (Math.random() - 0.5) * 3, 2 + Math.random() * 2, (Math.random() - 0.5) * 3, SPARK[lv], 0.25);
        }
      }
      if (kart.surface === 'dirt' && !kart.airborne && Math.abs(kart.speed) > 5) {
        emit(kart.x - fx * 1.2 + rx * (Math.random() - 0.5) * 1.6, kart.y + 0.3, kart.z - fz * 1.2 + rz * (Math.random() - 0.5) * 1.6,
          -fx * 2 + (Math.random() - 0.5) * 2, 1 + Math.random() * 2, -fz * 2 + (Math.random() - 0.5) * 2, DUST, 0.6);
      }
      for (let i = 0; i < MAX; i++) {
        const p = parts[i];
        if (p.life > 0) {
          p.life -= dt;
          p.vy -= 6 * dt;
          p.x += p.vx * dt;
          p.y += p.vy * dt;
          p.z += p.vz * dt;
        }
        pos[i * 3] = p.x;
        pos[i * 3 + 1] = p.life > 0 ? p.y : -100;
        pos[i * 3 + 2] = p.z;
        col[i * 3] = p.c.r;
        col[i * 3 + 1] = p.c.g;
        col[i * 3 + 2] = p.c.b;
      }
      geo.attributes.position.needsUpdate = true;
      geo.attributes.color.needsUpdate = true;
    },
  };
}
```

```js
// 遊べる時間（名前が KYUGO・SOKI のときだけ、1日20分まで）。M2 は端末の中で数える（M4 でサーバーにも記録する）
// 1日の区切りは日本時間の深夜2時（深夜2時から次の深夜2時までの24時間）
export const LIMITED_NAMES = ['KYUGO', 'SOKI'];
export const DEFAULT_LIMIT_MIN = 20;
const JST = 9 * 3600 * 1000;
const DAY_START = 2 * 3600 * 1000; // 深夜2時

// その時刻が属する「1日」の名前（日本時間の深夜2時で区切った日付。例 '2026-10-14'）
export function dayKey(nowMs) {
  const d = new Date(nowMs + JST - DAY_START);
  return d.toISOString().slice(0, 10);
}

export function isLimited(name) {
  return LIMITED_NAMES.includes(name);
}

// 使った秒数。store は { 日付: { 名前: 秒 } }
export function usedSec(store, name, nowMs) {
  return store[dayKey(nowMs)]?.[name] ?? 0;
}

// 残りの秒数。制限のない名前は Infinity
export function remainingSec(store, name, nowMs, limitMin = DEFAULT_LIMIT_MIN) {
  if (!isLimited(name)) return Infinity;
  return Math.max(0, limitMin * 60 - usedSec(store, name, nowMs));
}

// 遊んだ秒数を足す（制限のない名前は記録しない）。古い日の記録は消す
export function addPlay(store, name, nowMs, sec) {
  if (!isLimited(name) || sec <= 0) return store;
  const key = dayKey(nowMs);
  const today = { ...(store[key] ?? {}) };
  today[name] = (today[name] ?? 0) + sec;
  return { [key]: today };
}

// 画面に出す残り時間（1分を切ったら秒で）
export function remainingText(sec) {
  if (sec === Infinity) return '';
  if (sec < 60) return `Today: ${Math.ceil(sec)} sec left`;
  return `Today: ${Math.floor(sec / 60)} min left`;
}
```

```js
// 名前: アルファベットだけ、最大12文字、大文字。最近の名前は5件まで（最近のものが先頭）
export const NAME_MAX = 12;
export const RECENT_MAX = 5;

export function isValidName(name) {
  return typeof name === 'string' && /^[A-Z]{1,12}$/.test(name);
}

// 1文字足す（A〜Z だけ。12文字まで）
export function typeLetter(name, letter) {
  if (!/^[A-Z]$/.test(letter) || name.length >= NAME_MAX) return name;
```

```js
// レースの決まり: スタートの並び、カウント、スタートダッシュ、周回、順位
export const LAPS = 3; // 周回
export const COUNTDOWN = 3; // 3, 2, 1 のあと GO（秒）
export const DASH_FROM = -0.3; // GO の何秒前から押せばスタートダッシュになるか
export const DASH_TO = 0.1; // GO の何秒後まで押せばスタートダッシュになるか
export const GRID_GAP = 7; // スタートの並びの、前後の間（m）
export const GRID_SIDE = 4.5; // スタートの並びの、左右の間（中心線から、m）

// スタートの並び: 2列に並べる。k 番目の車の { i（点の番号）, lateral（中心線から右へ m） }
export function gridSlot(track, k) {
  const n = track.samples.length;
  const back = 5 + Math.floor(k / 2) * GRID_GAP;
  return { i: (track.startIndex - back + n) % n, lateral: k % 2 ? GRID_SIDE : -GRID_SIDE };
}

// t は GO からの秒（カウント中はマイナス）
export function createRace(count) {
  return {
    t: -COUNTDOWN,
    phase: 'countdown', // 'countdown' → 'racing'
    entries: Array.from({ length: count }, () => ({
      lap: 0, // 回り終えた周回の数
      rel: null, // スタートラインから何m進んだか（0〜1周の長さ）
      checkpoint: false, // その周の半分（コースの反対側）を通ったか
      finished: false,
      time: null, // ゴールした時刻（GO からの秒）
      pressedAt: null, // スタートダッシュのボタンを押した時刻
      dashDone: false,
    })),
  };
}

// 時間を進める。GO になったその回だけ true を返す
export function raceTick(race, dt) {
  race.t += dt;
  if (race.phase === 'countdown' && race.t >= 0) {
    race.phase = 'racing';
    return true;
  }
  return false;
}

// 画面の真ん中に出すカウント（'3' '2' '1' 'GO!'、出さないときは ''）
export function countdownText(race) {
  if (race.t < 0) return String(Math.ceil(-race.t));
  return race.t < 1 ? 'GO!' : '';
}

// スタートダッシュのボタン（DRIFT）を押した。最初に押した時刻だけを覚える
export function pressDash(race, k) {
  const e = race.entries[k];
  if (e.pressedAt === null && race.t <= DASH_TO) e.pressedAt = race.t;
}

// スタートダッシュになるかを、GO のあとで1回だけ決める。なるなら true
export function takeDash(race, k) {
  const e = race.entries[k];
  if (e.dashDone || race.t < 0) return false;
  if (e.pressedAt === null && race.t <= DASH_TO) return false; // まだ押すかもしれない
  e.dashDone = true;
  return e.pressedAt !== null && e.pressedAt >= DASH_FROM && e.pressedAt <= DASH_TO;
}

// 車がコースのどこにいるか（点の番号 seg）から、周回を数える
export function updateProgress(race, k, track, seg) {
  const e = race.entries[k];
  if (e.finished) return;
  const L = track.length;
  const start = track.samples[track.startIndex].s;
  const rel = (((track.samples[seg].s - start) % L) + L) % L;
  if (rel > 0.4 * L && rel < 0.6 * L) e.checkpoint = true;
  if (e.rel !== null) {
    if (e.rel > 0.8 * L && rel < 0.2 * L && e.checkpoint) {
      // スタートラインを前向きに越えた（その周の半分を通っていれば、1周）
      e.lap += 1;
      e.checkpoint = false;
      if (e.lap >= LAPS) {
        e.finished = true;
        e.time = race.t;
      }
    } else if (e.rel < 0.2 * L && rel > 0.8 * L && e.lap > 0 && !e.checkpoint) {
      // バックでスタートラインを越えて戻った
      e.lap -= 1;
      e.checkpoint = true;
    }
  }
  e.rel = rel;
}

// 進んだ距離（順位を決める。スタートラインの手前はマイナス）
export function progress(race, k, track) {
  const e = race.entries[k];
  const L = track.length;
  if (e.rel === null) return -Infinity;
  const rel = e.checkpoint || e.rel < L / 2 ? e.rel : e.rel - L;
  return e.lap * L + rel;
}

// 順位の順に並べた車の番号（ゴールした車はゴールした順、まだの車は進んだ距離の順）
export function ranking(race, track) {
  return race.entries
    .map((e, k) => k)
    .sort((a, b) => {
      const ea = race.entries[a];
      const eb = race.entries[b];
      if (ea.finished && eb.finished) return ea.time - eb.time;
      if (ea.finished) return -1;
      if (eb.finished) return 1;
      return progress(race, b, track) - progress(race, a, track);
    });
}

// 1st, 2nd, 3rd, 4th ...
export function ordinal(n) {
  const s = ['th', 'st', 'nd', 'rd'];
  const v = n % 100;
  return n + (s[(v - 20) % 10] || s[v] || s[0]);
}

// 0:23.45
export function timeText(sec) {
  const m = Math.floor(sec / 60);
  const s = sec - m * 60;
  return `${m}:${s < 10 ? '0' : ''}${s.toFixed(2)}`;
}
```

```js
// 車どうしの押し合い: 重なったら、おたがいに押し戻すだけ（はじき飛ばさない）
// 高さが離れている（ジャンプで上を飛んでいる）車とはぶつからない
// スピードが少し落ちるのは、ぶつかった最初の1回だけで、空中では落ちない（並んで走ったり飛んだりしても遅くならない）
// prev は前の回にぶつかっていた組（'a-b'）。今回ぶつかっている組を返す
export const KART_RADIUS = 1.1; // 車の大きさ（半径 m）
export const BUMP_KEEP = 0.98; // ぶつかったときに残る速さの割合

export function pushApart(karts, prev = new Set()) {
  const now = new Set();
  const min = KART_RADIUS * 2;
  for (let a = 0; a < karts.length; a++) {
    for (let b = a + 1; b < karts.length; b++) {
      const A = karts[a];
      const B = karts[b];
      if (A.respawn > 0 || B.respawn > 0 || Math.abs(A.y - B.y) > 1.5) continue;
      const dx = B.x - A.x;
      const dz = B.z - A.z;
      const d = Math.hypot(dx, dz);
      if (d >= min || d === 0) continue;
      const push = (min - d) / 2;
      const nx = dx / d;
      const nz = dz / d;
      A.x -= nx * push;
      A.z -= nz * push;
      B.x += nx * push;
      B.z += nz * push;
      const key = `${a}-${b}`;
      now.add(key);
      if (!prev.has(key) && !A.airborne && !B.airborne) {
        A.speed *= BUMP_KEEP;
        B.speed *= BUMP_KEEP;
      }
    }
  }
  return now;
}
```

```js
// CPU の運転: 道の中心から少しずれた線を、少し先を見て走る。きついカーブの手前ではブレーキ、
// 長いカーブではドリフト。プレイヤーより前の CPU は少し遅く、後ろの CPU は少し速く（離れすぎないように）
import { locate } from './track.js';

export const CPU = {
  LOOK: 10, // 何m先の点を目指すか（の基本）
  LOOK_PER_SPEED: 0.3, // 速さ1m/s ごとに、目指す点をこれだけ先にする
  STEER_GAIN: 2.5, // 目指す点とのずれ（rad）を、ハンドルの量にする倍率
  BEND_AHEAD: 35, // 何m先までのカーブを見るか
  BRAKE_BEND: 1.1, // この角度（rad）以上曲がるカーブの手前では、速すぎればブレーキ
  CORNER_SPEED: 25, // きついカーブに入る速さ（m/s）
  DRIFT_BEND: 0.6, // この角度以上曲がるカーブでは、ドリフトする
  DRIFT_END: 0.25, // カーブがこの角度より小さくなったら、ドリフトをやめる
  SKILL: 0.93, // CPU の最高速の割合（プレイヤーが「がんばれば勝てる」くらいに）
  RUBBER: 0.25, // プレイヤーとの差（1周に対する割合）を、速さの倍率にする強さ
  RUBBER_MAX: 0.08, // 速さを変えるのは、この割合まで
};

const wrap = (a) => Math.atan2(Math.sin(a), Math.cos(a));

// k 番目の CPU。走る線（中心から右へ m）と、線をゆらす速さを少しずつ変える
export function createCpu(k) {
  return {
    lane: [-5, 4, -2, 6, -6, 2, 5][k % 7],
    sway: 0.4 + (k % 3) * 0.15,
    phase: k * 1.7,
    drifting: false,
  };
}

// プレイヤーとの差から、速さの倍率を決める（progress の差、1周の長さ L）
export function rubberFactor(cpuProgress, playerProgress, L) {
  const diff = (playerProgress - cpuProgress) / L; // プレイヤーが前ならプラス
  return 1 + Math.max(-CPU.RUBBER_MAX, Math.min(CPU.RUBBER_MAX, diff * CPU.RUBBER));
}

// その時の操作（{ steer, brake, drift }）。t はレースの時刻（線をゆらすため）
export function cpuControl(cpu, kart, track, t) {
  const { samples } = track;
  const n = samples.length;
  const loc = locate(track, kart.x, kart.z, kart.seg);
  const look = Math.round(CPU.LOOK + Math.max(0, kart.speed) * CPU.LOOK_PER_SPEED);
  const p = samples[(loc.i + look) % n];
  const lane = cpu.lane + Math.sin(t * cpu.sway + cpu.phase) * 2;
  const tx = p.x - Math.sin(p.h) * lane;
  const tz = p.z + Math.cos(p.h) * lane;
  const angle = wrap(Math.atan2(tz - kart.z, tx - kart.x) - kart.move);
  const steer = Math.max(-1, Math.min(1, angle * CPU.STEER_GAIN));
  const bendTo = wrap(samples[(loc.i + CPU.BEND_AHEAD) % n].h - loc.heading);
  const bend = Math.abs(bendTo);
  const brake = bend > CPU.BRAKE_BEND && kart.speed > CPU.CORNER_SPEED;
  if (!cpu.drifting && bend > CPU.DRIFT_BEND && kart.speed > 20 && !kart.airborne) cpu.drifting = true;
  if (cpu.drifting && (bend < CPU.DRIFT_END || kart.speed < 14)) cpu.drifting = false;
  // ドリフトを始める（跳ねている）間だけ、曲がる向きにハンドルを切る。滑り出したら、目指す点に合わせて加減する
  const starting = cpu.drifting && !kart.drift.active;
  const out = starting ? Math.sign(bendTo) * Math.max(0.5, Math.abs(steer)) : steer;
  return { steer: out, brake, drift: cpu.drifting };
}

// CPU の車の数字（キャラの数字に、CPU の強さと、プレイヤーとの差の倍率をかける）
export function cpuParams(base, rubber) {
  return { ...base, topSpeed: base.topSpeed * CPU.SKILL * rubber };
}
```

```js
}
```

```js
      if (remainingSec(playStore(), name, Date.now()) <= 0) {
        await seeYouTomorrow(name);
        break;
      }
      if (pick !== 'again') break;
    }
  }
}

// ---- 描く ----
let frames = 0;
let fpsTime = 0;
let fps = 60;
let last = performance.now();
let orbit = 0;
const overHead = new THREE.Vector3();

// タイトルなどの画面のうしろでは、コースの上をゆっくり回って見せる
function orbitCamera(dt) {
  orbit += dt * 0.08;
  camera.fov = 60;
  camera.updateProjectionMatrix();
  camera.position.set(Math.sin(orbit) * 420, 160, Math.cos(orbit) * 420);
  camera.lookAt(0, 0, 0);
}

function renderGame(alpha, dt) {
  const g = game;
  const views = g.karts.map((kart, k) => {
    const cur = poseOf(kart);
    const prev = g.prev[k];
    const jumped = Math.hypot(cur.x - prev.x, cur.z - prev.z) > 10;
    return { ...kart, ...(jumped ? cur : blendPose(prev, cur, alpha)) };
  });
  views.forEach((v, k) => syncKartModel(g.models[k], v));
  const mine = views[g.me];
  const kart = g.karts[g.me];
  effects.update(mine, dt);
  if (dev.has('front')) updateFrontCamera(camera, mine);
  else updateChaseCamera(camera, mine, g.base.topSpeed, dt);
  speedLines.style.opacity = kart.boost > 0 ? 1 : 0;
  // 表示
  const { race } = g;
  const e = race.entries[g.me];
  speedNum.textContent = speedText(kart.speed);
  speedo.classList.toggle('boost', kart.boost > 0);
  lapEl.textContent = `LAP ${Math.min(LAPS, e.lap + 1)}/${LAPS}`;
  timeEl.textContent = timeText(Math.max(0, e.finished ? e.time : race.t));
  rankEl.textContent = ordinal(ranking(race, track).indexOf(g.me) + 1);
  countEl.textContent = e.finished ? 'FINISH!' : countdownText(race);
  const left = remainingSec(playStore(), g.name, Date.now()) - g.playSec;
  leftEl.textContent = remainingText(Math.max(0, left));
  return { mine, kart };
}

startLoop({
  update(dt) {
    controls.tick();
    if (game && !game.paused) stepGame(dt);
  },
  render(alpha) {
    const now = performance.now();
    const dt = Math.min(0.1, (now - last) / 1000);
    last = now;
    frames++;
    fpsTime += dt;
    if (fpsTime >= 2) {
      fps = Math.round(frames / fpsTime);
      frames = 0;
      fpsTime = 0;
      const q = nextQuality(quality, fps);
      if (q !== quality) {
        quality = q;
        resize();
      }
    }
    for (const n of BUTTONS) els[n].classList.toggle('active', controls.touch[n]);
    let wrong = false;
    let mine = null;
    if (game) {
      const r = renderGame(alpha, dt);
      mine = r.mine;
      wrong = r.kart.wrongWay;
    } else {
      orbitCamera(dt);
    }
    renderer.render(scene, camera);
    // 逆走の警告: キャラの頭の上の画面の位置に出す
    wrongWayEl.classList.toggle('show', wrong);
    if (wrong) {
      overHead.set(mine.x, mine.y + 3.6, mine.z).project(camera);
      wrongWayEl.style.left = `${((overHead.x + 1) / 2) * window.innerWidth}px`;
      wrongWayEl.style.top = `${((1 - overHead.y) / 2) * window.innerHeight}px`;
    }
    speedo.classList.toggle('hidden', !game);
    hud.textContent = `${fps}fps  ${controls.mode}  v:${VERSION}`;
  },
});

flow();
```

```js
// 音のおおもと: 最初のタップで AudioContext を作る、ON/OFF（端末に覚える）、全体の音量、裏に回ったら止める
// 音が使えない端末では ctx が null のまま。ほかの音の部品は、ctx がなければ何もしない
import { load, save } from '../storage.js';

export const MUSIC_VOLUME = 0.55; // 曲の音量（全体を 1 として）

function defaultContext() {
  const AC = window.AudioContext || window.webkitAudioContext;
  return new AC();
}

export function createAudio({ makeContext = defaultContext, store = { load, save } } = {}) {
  let on = store.load('sound', true) !== false;
  let ctx = null;
  let master = null;
  let musicOut = null;
  let sfxOut = null;
  let broken = false;

  function ensure() {
    if (ctx || broken) return ctx;
    try {
      ctx = makeContext();
      master = ctx.createGain();
      master.gain.value = on ? 1 : 0;
      master.connect(ctx.destination);
      musicOut = ctx.createGain();
      musicOut.gain.value = MUSIC_VOLUME;
      musicOut.connect(master);
      sfxOut = ctx.createGain();
      sfxOut.connect(master);
    } catch {
      broken = true;
      ctx = master = musicOut = sfxOut = null;
    }
    return ctx;
  }

  return {
    get ctx() {
      return ctx;
    },
    get master() {
      return master;
    },
    get musicOut() {
      return musicOut;
    },
    get sfxOut() {
      return sfxOut;
    },
    // 最初のタップ・クリック・キーのときに呼ぶ（iPhone・Android は、さわるまで音を出せない）
    unlock() {
      const c = ensure();
      if (c && c.state !== 'running') c.resume().catch(() => {});
    },
    isOn: () => on,
    setOn(value) {
      on = !!value;
      store.save('sound', on);
      if (master) master.gain.setTargetAtTime(on ? 1 : 0, ctx.currentTime, 0.02);
    },
    // 画面が見えなくなったとき・戻ったとき
    pause() {
      if (ctx) ctx.suspend().catch(() => {});
    },
    resume() {
      if (ctx) ctx.resume().catch(() => {});
    },
  };
}
```

```js
// 音の高さと、曲の書き方（楽譜の文字）を読む部品
const NAMES = { C: 0, D: 2, E: 4, F: 5, G: 7, A: 9, B: 11 };

// 'C4' 'F#5' 'Bb3' → MIDI の番号（C4 = 60）
export function noteToMidi(name) {
  const m = /^([A-G])(#|b)?(-?\d)$/.exec(name);
  if (!m) throw new Error(`音の名前がおかしい: ${name}`);
  const acc = m[2] === '#' ? 1 : m[2] === 'b' ? -1 : 0;
  return 12 * (Number(m[3]) + 1) + NAMES[m[1]] + acc;
}

// MIDI の番号 → 周波数（Hz）。A4（69）= 440Hz
export function midiToHz(midi) {
  return 440 * 2 ** ((midi - 69) / 12);
}

// 'E5:2 G5:2 R:4' → { notes: [{ step, len, midi }], steps }（長さは16分音符の数。R は休み）
export function parseTrack(text) {
  const notes = [];
  let step = 0;
  for (const tok of text.trim().split(/\s+/)) {
    const [name, lenText] = tok.split(':');
    const len = Number(lenText);
    if (!Number.isInteger(len) || len <= 0) throw new Error(`長さがおかしい: ${tok}`);
    if (name !== 'R') notes.push({ step, len, midi: noteToMidi(name) });
    step += len;
  }
  return { notes, steps: step };
}

// コードの名前（'C' 'Am' 'F#m' 'Bb'）→ 3つの音（根音は octave の高さ）
export function chordNotes(name, octave = 4) {
  const m = /^([A-G](#|b)?)(m?)$/.exec(name);
  if (!m) throw new Error(`コードの名前がおかしい: ${name}`);
  const root = noteToMidi(`${m[1]}${octave}`);
  return [root, root + (m[3] ? 3 : 4), root + 7];
}

// 1小節に1つか2つのコード（'F G' は前半が F、後半が G）→ [{ step, len, name }]
export function chordSteps(chords) {
  const out = [];
  chords.forEach((bar, i) => {
    const parts = bar.split(' ');
    const len = 16 / parts.length;
    parts.forEach((name, k) => out.push({ step: i * 16 + k * len, len, name }));
  });
  return out;
}
```

```js
// 曲: 楽譜のデータから音の並びを作り（テストできる）、OfflineAudioContext で一度だけ音にして、ループで流す
import { parseTrack, chordNotes, chordSteps } from './notes.js';

// 16分音符1つの秒数と、曲1回分の秒数
export const stepSec = (song) => 60 / song.bpm / 4;
export const loopSec = (song) => song.bars * 16 * stepSec(song);

// 曲の音の並び: [{ t（秒）, dur（秒）, inst, midi? }]
export function songEvents(song) {
  const s = stepSec(song);
  const ev = [];
  for (const n of parseTrack(song.lead).notes) ev.push({ t: n.step * s, dur: n.len * s * 0.9, inst: 'lead', midi: n.midi });
  for (const c of chordSteps(song.chords)) {
    // ハモリ（コードの3つの音を長くのばす）
    for (const midi of chordNotes(c.name, 4)) ev.push({ t: c.step * s, dur: c.len * s, inst: 'pad', midi });
    // ベース: 8分音符で、根音を低い・高いと交互に
    const root = chordNotes(c.name, 2)[0];
    for (let k = 0; k < c.len; k += 2) ev.push({ t: (c.step + k) * s, dur: s * 1.8, inst: 'bass', midi: root + ((k / 2) % 2 ? 12 : 0) });
  }
  for (let bar = 0; bar < song.bars; bar++) {
    for (const inst of ['kick', 'snare', 'hat']) {
      [...song.drums[inst]].forEach((c, k) => {
        if (c === 'x') ev.push({ t: (bar * 16 + k) * s, dur: s, inst });
      });
    }
  }
  return ev;
}

// 残響の余り（loopLen より後ろ）を曲の頭に重ねて、loopLen の長さにする（ループのつなぎ目で残響が切れない）
export function wrapTail(channels, loopLen) {
  return channels.map((data) => {
    const out = data.slice(0, loopLen);
    for (let i = loopLen; i < data.length; i++) out[(i - loopLen) % loopLen] += data[i];
    return out;
  });
}
```

```js
    else if (cur.count === 'GO!') ev.push('go');
```

```js
export function createSfx(audio) {
  let noise = null; // 1秒の雑音（使い回す）

  function noiseBuffer(ctx) {
    if (!noise) {
      noise = ctx.createBuffer(1, ctx.sampleRate, ctx.sampleRate);
      const d = noise.getChannelData(0);
      for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1;
    }
    return noise;
  }

  // 短い音: type の波で、freq から to まで高さを変えながら、dur 秒で消える
  function tone(type, freq, dur, gain, { to = freq, at = 0 } = {}) {
    const ctx = audio.ctx;
    const t = ctx.currentTime + at;
    const o = ctx.createOscillator();
    o.type = type;
    o.frequency.setValueAtTime(freq, t);
    if (to !== freq) o.frequency.exponentialRampToValueAtTime(to, t + dur);
    const g = ctx.createGain();
    g.gain.setValueAtTime(gain, t);
    g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
    o.connect(g).connect(audio.sfxOut);
    o.start(t);
    o.stop(t + dur + 0.05);
  }

  // 短い雑音: フィルター（type）の周波数を from から to へ動かす
  function burst(type, from, to, dur, gain, { at = 0 } = {}) {
    const ctx = audio.ctx;
    const t = ctx.currentTime + at;
    const src = ctx.createBufferSource();
    src.buffer = noiseBuffer(ctx);
    const f = ctx.createBiquadFilter();
    f.type = type;
    f.frequency.setValueAtTime(from, t);
    f.frequency.exponentialRampToValueAtTime(to, t + dur);
    const g = ctx.createGain();
    g.gain.setValueAtTime(gain, t);
    g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
    src.connect(f).connect(g).connect(audio.sfxOut);
    src.start(t);
    src.stop(t + dur + 0.05);
  }

    midis.forEach((m, i) => {
      tone('square', midiToHz(m), dur, gain, { at: i * gap });
      tone('triangle', midiToHz(m + 12), dur, gain * 0.5, { at: i * gap });

  return {
    play(name) {
      if (audio.ctx && SOUNDS[name]) SOUNDS[name]();
    },
    engine: {
      start() {
        const ctx = audio.ctx;
        if (!ctx || engine) return;
        const filter = ctx.createBiquadFilter();
        filter.type = 'lowpass';
        filter.frequency.value = 900;
        const gain = ctx.createGain();
        gain.gain.value = 0;
        filter.connect(gain).connect(audio.sfxOut);
        const oscs = [-8, 8].map((detune) => {
          const o = ctx.createOscillator();
          o.type = 'sawtooth';
          o.frequency.value = 55;
          o.detune.value = detune;
          o.connect(filter);
          o.start();
          return o;
        });
        engine = { oscs, filter, gain };
      },
      // 毎フレーム呼ぶ（値をなめらかに近づけるだけ。物は作らない）
      update(speed, top, boosting) {
        if (!engine) return;
        const p = engineParams(speed, top, boosting);
        const t = audio.ctx.currentTime;
        for (const o of engine.oscs) o.frequency.setTargetAtTime(p.freq, t, 0.05);
        engine.gain.gain.setTargetAtTime(p.gain, t, 0.08);
      },
      stop() {
        if (!engine) return;
        for (const o of engine.oscs) o.stop();
        engine.gain.disconnect();
        engine = null;
      },
    },
    // ドリフトのキュルキュル（on のあいだ鳴る。speed01 は 0〜1 の速さ）
    drift(on, speed01 = 1) {
      const ctx = audio.ctx;
      if (!ctx) return;
      if (on && !squeal) {
        const src = ctx.createBufferSource();
        src.buffer = noiseBuffer(ctx);
        src.loop = true;
        const filter = ctx.createBiquadFilter();
        filter.type = 'bandpass';
        filter.frequency.value = 2600;
        filter.Q.value = 6;
        const gain = ctx.createGain();
        gain.gain.value = 0;
        src.connect(filter).connect(gain).connect(audio.sfxOut);
        src.start();
        squeal = { src, gain };
      }
      if (squeal) squeal.gain.gain.setTargetAtTime(on ? 0.12 * speed01 : 0, ctx.currentTime, 0.05);
      if (!on && squeal) {
        const s = squeal;
        squeal = null;
        s.src.stop(ctx.currentTime + 0.2);
        setTimeout(() => s.gain.disconnect(), 300);
      }
    },
    // タイトルへ戻る・一時停止・次のレースのとき: 鳴りつづける音を全部止める
    stopAll() {
      this.engine.stop();
      this.drift(false);
    },
  };
}
```

```js
export const CROWD_MAX = 0.35; // 歓声のいちばん大きい音量
const CROWD_NEAR = 25; // ここまで近いと、いちばん大きい（m）
const CROWD_FAR = 200; // ここより遠いと、聞こえない（m）

// 歓声: 観客席（spot）との距離で大きさ、車の右か左かで左右（-1 左 〜 1 右）
export function crowdMix(kart, spot) {
  const dx = spot.x - kart.x;
  const dz = spot.z - kart.z;
  const d = Math.hypot(dx, dz);
  const near = Math.max(0, Math.min(1, 1 - (d - CROWD_NEAR) / (CROWD_FAR - CROWD_NEAR)));
  // 右向き（コースの横の向きと同じ決め方。heading は x から z へ回る角度）
  const rx = -Math.sin(kart.heading);
  const rz = Math.cos(kart.heading);
  const pan = d < 1e-6 ? 0 : Math.max(-1, Math.min(1, (dx * rx + dz * rz) / d));
  return { gain: CROWD_MAX * near * near, pan };
}
```

```js
// 観客席の向き（コースの外側）と、真ん中の場所。音（歓声）でも使う
function grandstandFrame(track) {
  const p = track.samples[track.startIndex];
  const base = p.wall + 4;
  const [r0x, r0z] = rightOf(p);
  const side = insideLoop(track.samples, p.x + r0x * (base + 6), p.z + r0z * (base + 6)) ? -1 : 1;
  return { p, base, rx: r0x * side, rz: r0z * side };
}

export function grandstandSpot(track) {
  const { p, base, rx, rz } = grandstandFrame(track);
  const mid = base + 9; // 10段（2m ずつ）の真ん中
  return { x: p.x + rx * mid, z: p.z + rz * mid };
}
```

```js
// 観客の歓声: 雑音を何層も重ねた「ワーッ」をループで流し、ときどき「ウォー！」と大きくなる
// 大きさと左右は update で毎フレーム近づける（物は作らない）
export function createCrowd(audio) {
  let node = null; // { src, pan, gain, swell }
  let buffer = null;
  let boostUntil = 0;

  // 4秒の歓声（ループ）: 帯域を変えた雑音を3つ重ね、ゆっくり大きさをゆらす
  function makeBuffer(ctx) {
    const len = ctx.sampleRate * 4;
    const b = ctx.createBuffer(2, len, ctx.sampleRate);
    for (let ch = 0; ch < 2; ch++) {
      const d = b.getChannelData(ch);
      let low = 0;
      let mid = 0;
      for (let i = 0; i < len; i++) {
        const w = Math.random() * 2 - 1;
        low += (w - low) * 0.02; // こもった「ワー」
        mid += (w - mid) * 0.15; // 少し明るい声のざわめき
        const t = i / ctx.sampleRate;
        // ゆれ（4秒で何回かぴったり戻る速さにして、ループのつなぎ目でとぎれない）
        const sway = 0.7 + 0.3 * Math.sin((2 * Math.PI * t * 3) / 4 + ch);
        const cheer = 0.5 + 0.5 * Math.max(0, Math.sin((2 * Math.PI * t) / 4)) ** 4; // 4秒に1回「ウォー！」
        d[i] = (low * 3 + mid * 0.8) * sway * cheer;
      }
    }
    return b;
  }

  return {
    start() {
      const ctx = audio.ctx;
      if (!ctx || node) return;
      buffer = buffer || makeBuffer(ctx);
      const src = ctx.createBufferSource();
      src.buffer = buffer;
      src.loop = true;
      const pan = ctx.createStereoPanner();
      const gain = ctx.createGain();
      gain.gain.value = 0;
      src.connect(gain).connect(pan).connect(audio.sfxOut);
      src.start();
      node = { src, pan, gain };
    },
    update({ gain, pan }) {
      if (!node) return;
      const t = audio.ctx.currentTime;
      const g = t < boostUntil ? Math.min(1, gain * 2 + 0.15) : gain;
      node.gain.gain.setTargetAtTime(g, t, 0.15);
      node.pan.pan.setTargetAtTime(pan, t, 0.15);
    },
    // 最後の周・ゴールのとき、しばらく大きく
    excite(seconds) {
      if (audio.ctx) boostUntil = audio.ctx.currentTime + seconds;
    },
    stop() {
      if (!node) return;
      node.src.stop();
      node.pan.disconnect();
      node = null;
    },
  };
}
```

```js
import { midiToHz } from './notes.js';
import { SONGS } from './songs.js';

export const FINAL_LAP_RATE = 1.12; // 最後の周は、曲をこの速さで流す
const REVERB_SEC = 1.5; // 残響の長さ

function defaultOffline(channels, length, sampleRate) {
  const OAC = window.OfflineAudioContext || window.webkitOfflineAudioContext;
  return new OAC(channels, length, sampleRate);
}

// 曲を音にする（ゲームを開いたあとに一度だけ）。作れない端末では null
export async function renderSong(song, { sampleRate = 22050, makeOffline = defaultOffline } = {}) {
  try {
    const loopLen = Math.round(loopSec(song) * sampleRate);
    const ctx = makeOffline(2, loopLen + Math.round(REVERB_SEC * sampleRate), sampleRate);
    const out = ctx.createDynamicsCompressor();
    out.connect(ctx.destination);
    const dry = ctx.createGain();
    dry.gain.value = 0.8;
    dry.connect(out);
    // 残響: 雑音がだんだん小さくなる響き
    const verb = ctx.createConvolver();
    const ir = ctx.createBuffer(2, Math.round(REVERB_SEC * sampleRate), sampleRate);
    for (let ch = 0; ch < 2; ch++) {
      const d = ir.getChannelData(ch);
      for (let i = 0; i < d.length; i++) d[i] = (Math.random() * 2 - 1) * (1 - i / d.length) ** 3;
    }
    verb.buffer = ir;
    const wet = ctx.createGain();
    wet.gain.value = 0.2;
    dry.connect(verb).connect(wet).connect(out);
    const noise = ctx.createBuffer(1, sampleRate, sampleRate);
    const nd = noise.getChannelData(0);
    for (let i = 0; i < nd.length; i++) nd[i] = Math.random() * 2 - 1;

    for (const e of songEvents(song)) playEvent(ctx, dry, noise, e);
    const rendered = await ctx.startRendering();
    const chans = wrapTail([rendered.getChannelData(0), rendered.getChannelData(1)], loopLen);
    const buf = ctx.createBuffer(2, loopLen, sampleRate);
    chans.forEach((d, ch) => buf.getChannelData(ch).set(d));
    return buf;
  } catch {
    return null;
  }
}

// 音の高さのある楽器: 波の形、フィルター、音量の山（アタック → のばす → 消える）
function voice(ctx, dest, { type, freq, t, dur, gain, detune = 0, cutoff = 0, attack = 0.01, release = 0.08 }) {
  const o = ctx.createOscillator();
  o.type = type;
  o.frequency.value = freq;
  o.detune.value = detune;
  const g = ctx.createGain();
  g.gain.setValueAtTime(0, t);
  g.gain.linearRampToValueAtTime(gain, t + attack);
  g.gain.setTargetAtTime(gain * 0.7, t + attack, dur * 0.3);
  g.gain.setTargetAtTime(0, t + dur, release / 3);
  let node = o;
  if (cutoff) {
    const f = ctx.createBiquadFilter();
    f.type = 'lowpass';
    f.frequency.value = cutoff;
    o.connect(f);
    node = f;
  }
  node.connect(g).connect(dest);
  o.start(t);
  o.stop(t + dur + release * 3);
}

// 雑音の楽器（スネア・ハイハット）
function hit(ctx, dest, noise, { t, cutoff, gain, decay }) {
  const s = ctx.createBufferSource();
  s.buffer = noise;
  const f = ctx.createBiquadFilter();
  f.type = 'highpass';
  f.frequency.value = cutoff;
  const g = ctx.createGain();
  g.gain.setValueAtTime(gain, t);
  g.gain.exponentialRampToValueAtTime(0.0001, t + decay);
  s.connect(f).connect(g).connect(dest);
  s.start(t);
  s.stop(t + decay + 0.02);
}

function playEvent(ctx, dest, noise, e) {
  const { t, dur } = e;
  if (e.inst === 'lead') {
    // 少しだけ高さをずらした2つの音を重ねて厚みを出す
    for (const detune of [-8, 8]) voice(ctx, dest, { type: 'square', freq: midiToHz(e.midi), t, dur, gain: 0.07, detune, cutoff: 3000 });
  } else if (e.inst === 'pad') {
    voice(ctx, dest, { type: 'triangle', freq: midiToHz(e.midi), t, dur, gain: 0.035, attack: 0.08, release: 0.3 });
  } else if (e.inst === 'bass') {
    voice(ctx, dest, { type: 'square', freq: midiToHz(e.midi), t, dur, gain: 0.13, cutoff: 700 });
  } else if (e.inst === 'kick') {
    const o = ctx.createOscillator();
    o.type = 'sine';
    o.frequency.setValueAtTime(150, t);
    o.frequency.exponentialRampToValueAtTime(45, t + 0.12);
    const g = ctx.createGain();
    g.gain.setValueAtTime(0.7, t);
    g.gain.exponentialRampToValueAtTime(0.0001, t + 0.25);
    o.connect(g).connect(dest);
    o.start(t);
    o.stop(t + 0.3);
  } else if (e.inst === 'snare') {
    hit(ctx, dest, noise, { t, cutoff: 1800, gain: 0.3, decay: 0.14 });
    voice(ctx, dest, { type: 'triangle', freq: 190, t, dur: 0.05, gain: 0.15 });
  } else if (e.inst === 'hat') {
    hit(ctx, dest, noise, { t, cutoff: 7000, gain: 0.08, decay: 0.035 });
  }
}

// 曲を流す: prepare で作っておき（一度だけ）、play でループ、setRate で速さ、stop でだんだん消す
export function createMusic(audio, { render = renderSong } = {}) {
  const made = new Map(); // 曲の名前 → Promise<AudioBuffer | null>
  let current = null; // { name, src, gain }
  let token = 0;

  function prepare(name) {
    if (!made.has(name)) made.set(name, render(SONGS[name]));
    return made.get(name);
  }

  function stop(fade = 0.4) {
    token++;
    if (!current || !audio.ctx) {
      current = null;
      return;
    }
    const { src, gain } = current;
    const t = audio.ctx.currentTime;
    gain.gain.setTargetAtTime(0, t, fade / 3);
    src.stop(t + fade + 0.1);
    current = null;
  }

  async function play(name) {
    if (current && current.name === name) return;
    stop();
    const my = token;
    const buf = await prepare(name);
    if (my !== token || !buf || !audio.ctx || !audio.musicOut) return; // 待っている間に別の曲に変わった、または作れなかった
    const ctx = audio.ctx;
    const src = ctx.createBufferSource();
    src.buffer = buf;
    src.loop = true;
    const gain = ctx.createGain();
    gain.gain.setValueAtTime(0, ctx.currentTime);
    gain.gain.setTargetAtTime(1, ctx.currentTime, 0.1);
    src.connect(gain).connect(audio.musicOut);
    src.start();
    current = { name, src, gain };
  }

  function setRate(rate) {
    if (current && audio.ctx) current.src.playbackRate.setTargetAtTime(rate, audio.ctx.currentTime, 0.3);
  }

  return { prepare, play, setRate, stop };
}
```

```js
function loadSw({ online }) {
  const handlers = {};
  const cache = new Map();
  const self = { addEventListener: (type, fn) => (handlers[type] = fn), skipWaiting() {}, clients: { claim() {} } };
  const caches = {
    open: async () => ({ put: async (req, res) => cache.set(req.url, res) }),
    match: async (req) => cache.get(req.url),
  };
  const fetch = async (req) => {
    if (!online) throw new Error('offline');
    return { body: `new ${req.url}`, clone() { return this; } };
  };
  const location = { origin: 'https://example.com' };
  new Function('self', 'caches', 'fetch', 'location', read('sw.js').toString())(self, caches, fetch, location);
  const respond = async (url, method = 'GET') => {
    let p = null;
    handlers.fetch({ request: { url, method }, respondWith: (x) => (p = x) });
    return p && (await p);
  };
  return { respond, cache };
}
```

```js
const CACHE = 'ks-kart-dev'; // 公開のとき scripts/deploy.sh がコミットの番号に置き換える
```
